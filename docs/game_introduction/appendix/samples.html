game_loop.py

 1"""ゲームループの最小の例（4 章）。
 2
 3落ちてくる果物をかごで受け止める端末上のゲーム。
 41 回のループで「入力 → 更新 → 描画」を順に行う。
 5入力を待つ手番制のループなので、1 回の入力が 1 フレームに当たる。
 6
 7実行方法: py game_loop.py
 8操作: a で左、d で右、何も入力せずに Enter で待機、q で終了
 9"""
10
11import random
12from dataclasses import dataclass
13
14WIDTH = 7   # 横のマス数
15HEIGHT = 6  # 縦のマス数
16LIVES = 3   # 受け止めそこねてよい回数
17
18
19@dataclass
20class State:
21    """ゲームの状態。ループのたびに更新される。"""
22
23    basket_x: int
24    fruit_x: int
25    fruit_y: int
26    score: int = 0
27    misses: int = 0
28
29
30def read_input() -> str:
31    """入力: プレイヤーの操作を 1 つ読む。"""
32    return input("a/d/Enter/q > ").strip().lower()
33
34
35def update(state: State, command: str) -> None:
36    """更新: 操作とルールに従って状態を 1 段階進める。"""
37    if command == "a":
38        state.basket_x = max(0, state.basket_x - 1)
39    elif command == "d":
40        state.basket_x = min(WIDTH - 1, state.basket_x + 1)
41    state.fruit_y += 1
42    if state.fruit_y == HEIGHT - 1:  # 果物がかごの高さに着いた
43        if state.fruit_x == state.basket_x:
44            state.score += 1
45        else:
46            state.misses += 1
47        state.fruit_x = random.randrange(WIDTH)
48        state.fruit_y = 0
49
50
51def draw(state: State) -> None:
52    """描画: 状態を画面（端末）に表示する。"""
53    for y in range(HEIGHT):
54        row = ["."] * WIDTH
55        if y == state.fruit_y:
56            row[state.fruit_x] = "o"
57        if y == HEIGHT - 1:
58            row[state.basket_x] = "U"
59        print("".join(row))
60    print(f"score={state.score} misses={state.misses}/{LIVES}")
61
62
63def main() -> None:
64    state = State(basket_x=WIDTH // 2, fruit_x=random.randrange(WIDTH), fruit_y=0)
65    draw(state)
66    while state.misses < LIVES:
67        command = read_input()
68        if command == "q":
69            break
70        update(state, command)
71        draw(state)
72    print(f"game over: score={state.score}")
73
74
75if __name__ == "__main__":
76    main()

tictactoe.py

  1"""三目並べの局面を数え、ミニマックス法の AI と対戦する（5 章、8 章）。
  2
  3実行方法:
  4    py tictactoe.py          局面数とゲームの値を表示する
  5    py tictactoe.py play     AI と対戦する（あなたが先手の X）
  6"""
  7
  8import sys
  9from functools import cache
 10
 11Board = str  # 9 文字の文字列。"X"、"O"、" " で 3×3 のマスを左上から表す
 12
 13EMPTY: Board = " " * 9
 14LINES = [
 15    (0, 1, 2), (3, 4, 5), (6, 7, 8),  # 横
 16    (0, 3, 6), (1, 4, 7), (2, 5, 8),  # 縦
 17    (0, 4, 8), (2, 4, 6),             # 斜め
 18]
 19
 20
 21def winner(board: Board) -> str | None:
 22    """勝者（"X" か "O"）を返す。まだ決まっていなければ None を返す。"""
 23    for a, b, c in LINES:
 24        if board[a] != " " and board[a] == board[b] == board[c]:
 25            return board[a]
 26    return None
 27
 28
 29def is_over(board: Board) -> bool:
 30    return winner(board) is not None or " " not in board
 31
 32
 33def turn(board: Board) -> str:
 34    """次に打つ側を返す。X が先手。"""
 35    return "X" if board.count("X") == board.count("O") else "O"
 36
 37
 38def moves(board: Board) -> list[int]:
 39    return [i for i, cell in enumerate(board) if cell == " "]
 40
 41
 42def play(board: Board, square: int) -> Board:
 43    return board[:square] + turn(board) + board[square + 1:]
 44
 45
 46def count_games(board: Board = EMPTY) -> int:
 47    """終局までの手順の数（ゲーム木の葉の数）を数える。"""
 48    if is_over(board):
 49        return 1
 50    return sum(count_games(play(board, m)) for m in moves(board))
 51
 52
 53def reachable_positions(board: Board = EMPTY) -> set[Board]:
 54    """初期局面から到達できる局面の集合を返す（初期局面を含む）。"""
 55    found = {board}
 56    stack = [board]
 57    while stack:
 58        current = stack.pop()
 59        if is_over(current):
 60            continue
 61        for m in moves(current):
 62            nxt = play(current, m)
 63            if nxt not in found:
 64                found.add(nxt)
 65                stack.append(nxt)
 66    return found
 67
 68
 69@cache
 70def value(board: Board) -> int:
 71    """ミニマックス法による局面の値。X の勝ちが +1、引き分けが 0、O の勝ちが -1。"""
 72    w = winner(board)
 73    if w is not None:
 74        return 1 if w == "X" else -1
 75    if " " not in board:
 76        return 0
 77    values = [value(play(board, m)) for m in moves(board)]
 78    return max(values) if turn(board) == "X" else min(values)
 79
 80
 81def best_move(board: Board) -> int:
 82    """手番の側にとって最もよい手を返す。"""
 83    pick = max if turn(board) == "X" else min
 84    return pick(moves(board), key=lambda m: value(play(board, m)))
 85
 86
 87def show(board: Board) -> None:
 88    cells = [c if c != " " else str(i) for i, c in enumerate(board)]
 89    for row in range(3):
 90        print(" " + " | ".join(cells[row * 3:row * 3 + 3]))
 91
 92
 93def play_against_ai() -> None:
 94    board = EMPTY
 95    while not is_over(board):
 96        show(board)
 97        if turn(board) == "X":
 98            text = input("マスの番号 > ").strip()
 99            if not text.isdigit() or int(text) not in moves(board):
100                print("空いているマスの番号を入力する")
101                continue
102            board = play(board, int(text))
103        else:
104            board = play(board, best_move(board))
105    show(board)
106    print(f"勝者: {winner(board) or '引き分け'}")
107
108
109def main() -> None:
110    if len(sys.argv) > 1 and sys.argv[1] == "play":
111        play_against_ai()
112        return
113    positions = reachable_positions()
114    print(f"終局までの手順の数: {count_games()}")
115    print(f"到達できる局面の数: {len(positions)}")
116    print(f"初期局面の値: {value(EMPTY)}（0 は双方が最善を尽くすと引き分け）")
117
118
119if __name__ == "__main__":
120    main()

nim.py

 1"""ニムの必勝法を使う AI と対戦する（5 章、8 章）。
 2
 3ルール: 2 人が交互に、1 つの山から 1 個以上の石を取る。最後の石を取った方が勝ち。
 4
 5実行方法:
 6    py nim.py            山 3, 4, 5 で対戦する（あなたが先手）
 7    py nim.py 1 3 5 7    山の大きさを指定して対戦する
 8"""
 9
10import random
11import sys
12from functools import reduce
13from operator import xor
14
15
16def nim_sum(heaps: list[int]) -> int:
17    """すべての山の石の数の排他的論理和（ニム和）を返す。"""
18    return reduce(xor, heaps, 0)
19
20
21def winning_move(heaps: list[int]) -> tuple[int, int] | None:
22    """ニム和を 0 にする手（山の番号, 取る個数）を返す。無ければ None を返す。"""
23    s = nim_sum(heaps)
24    if s == 0:
25        return None
26    for i, h in enumerate(heaps):
27        target = h ^ s
28        if target < h:
29            return i, h - target
30    return None  # s != 0 なら必ず上で見つかるので、ここには来ない
31
32
33def ai_move(heaps: list[int]) -> tuple[int, int]:
34    """必勝の手があればそれを選び、無ければ 1 個だけ取って相手の失敗を待つ。"""
35    move = winning_move(heaps)
36    if move is not None:
37        return move
38    i = random.choice([i for i, h in enumerate(heaps) if h > 0])
39    return i, 1
40
41
42def read_move(heaps: list[int]) -> tuple[int, int]:
43    while True:
44        parts = input("山の番号と取る個数 > ").split()
45        if len(parts) == 2 and all(p.isdigit() for p in parts):
46            i, n = int(parts[0]), int(parts[1])
47            if 0 <= i < len(heaps) and 1 <= n <= heaps[i]:
48                return i, n
49        print("例: 0 2（0 番の山から 2 個取る）")
50
51
52def main() -> None:
53    heaps = [int(a) for a in sys.argv[1:]] or [3, 4, 5]
54    player_turn = True
55    while sum(heaps) > 0:
56        print("山: " + "  ".join(f"[{i}] {h}" for i, h in enumerate(heaps)) + f"  （ニム和 {nim_sum(heaps)}）")
57        if player_turn:
58            i, n = read_move(heaps)
59        else:
60            i, n = ai_move(heaps)
61            print(f"AI: {i} 番の山から {n} 個取る")
62        heaps[i] -= n
63        player_turn = not player_turn
64    print("あなたの勝ち" if not player_turn else "AI の勝ち")
65
66
67if __name__ == "__main__":
68    main()

rps.py

 1"""勝ったときの得点が手ごとに違うじゃんけんの最適な混合戦略を求める（5 章、8 章）。
 2
 3例として、グーで勝つと 3 歩、チョキとパーで勝つと 6 歩進む遊び（グリコ）を扱う。
 4得点は相手との差（ゼロ和）とし、相手がどの手を出しても期待値が同じになる手の割合を求める。
 5
 6実行方法:
 7    py rps.py            グリコの得点で計算する
 8    py rps.py 1 1 1      グー、パー、チョキで勝ったときの得点を指定する
 9"""
10
11import random
12import sys
13
14import numpy as np
15import numpy.typing as npt
16
17HANDS = ["グー", "パー", "チョキ"]
18
19
20def payoff_matrix(points: tuple[float, float, float]) -> npt.NDArray[np.float64]:
21    """行の手を出した側が得る点（相手との差）の表を返す。
22
23    points はグー、パー、チョキで勝ったときの得点。
24    """
25    rock, paper, scissors = points
26    return np.array([
27        [0.0, -paper, rock],        # グー: パーに負け、チョキに勝つ
28        [paper, 0.0, -scissors],    # パー: グーに勝ち、チョキに負ける
29        [-rock, scissors, 0.0],     # チョキ: グーに負け、パーに勝つ
30    ])
31
32
33def equilibrium(points: tuple[float, float, float]) -> npt.NDArray[np.float64]:
34    """どの手の期待値も 0 になる混合戦略（ナッシュ均衡）を求める。
35
36    A p = 0 と p の和 = 1 を連立して最小二乗法で解く。
37    """
38    a = payoff_matrix(points)
39    system = np.vstack([a, np.ones(3)])
40    rhs = np.array([0.0, 0.0, 0.0, 1.0])
41    solution, *_ = np.linalg.lstsq(system, rhs, rcond=None)
42    return np.asarray(solution, dtype=np.float64)
43
44
45def simulate(points: tuple[float, float, float], mine: list[float], theirs: list[float], rounds: int,
46             seed: int = 1) -> float:
47    """2 つの混合戦略を対戦させ、1 回あたりの平均得点を返す。"""
48    rng = random.Random(seed)
49    a = payoff_matrix(points)
50    total = 0.0
51    for _ in range(rounds):
52        i = rng.choices(range(3), weights=mine)[0]
53        j = rng.choices(range(3), weights=theirs)[0]
54        total += float(a[i, j])
55    return total / rounds
56
57
58def main() -> None:
59    values = [float(v) for v in sys.argv[1:4]] or [3.0, 6.0, 6.0]
60    points = (values[0], values[1], values[2])
61    p = equilibrium(points)
62    print("勝ったときの得点: " + "、".join(f"{h} {v:g}" for h, v in zip(HANDS, points)))
63    print("最適な割合:       " + "、".join(f"{h} {x:.3f}" for h, x in zip(HANDS, p)))
64    a = payoff_matrix(points)
65    uniform = np.full(3, 1 / 3)
66    print("最適な割合の相手に対する各手の期待値: " + "、".join(f"{h} {round(v, 9) + 0.0:+.3f}" for h, v in zip(HANDS, a @ p)))
67    print("均等に出す相手に対する各手の期待値:   " + "、".join(f"{h} {round(v, 9) + 0.0:+.3f}" for h, v in zip(HANDS, a @ uniform)))
68    best = int(np.argmax(a @ uniform))
69    mine = [1.0 if i == best else 0.0 for i in range(3)]
70    result = simulate(points, mine, list(uniform), 100_000)
71    print(f"均等に出す相手に毎回{HANDS[best]}を出したときの平均得点（シミュレーション）: {result:+.3f}")
72
73
74if __name__ == "__main__":
75    main()

dice.py

 1"""ダイスの和の分布を計算し、実際に振った結果と比べる（5 章）。
 2
 3実行方法:
 4    py dice.py           1d6、2d6、3d6 の分布、期待値、分散を表示する
 5    py dice.py 2 1000    2d6 を 1000 回振った結果を理論値と比べる
 6"""
 7
 8import random
 9import sys
10from collections import Counter
11from fractions import Fraction
12
13
14def sum_distribution(dice: int, faces: int = 6) -> dict[int, Fraction]:
15    """dice 個の faces 面ダイスの和が各値になる確率を返す。
16
17    1 個ずつ足していく（畳み込み）ので、全組合せを列挙するより速い。
18    """
19    dist: dict[int, Fraction] = {0: Fraction(1)}
20    for _ in range(dice):
21        nxt: dict[int, Fraction] = {}
22        for total, p in dist.items():
23            for face in range(1, faces + 1):
24                nxt[total + face] = nxt.get(total + face, Fraction(0)) + p / faces
25        dist = nxt
26    return dist
27
28
29def mean(dist: dict[int, Fraction]) -> Fraction:
30    return sum((v * p for v, p in dist.items()), Fraction(0))
31
32
33def variance(dist: dict[int, Fraction]) -> Fraction:
34    m = mean(dist)
35    return sum(((v - m) ** 2 * p for v, p in dist.items()), Fraction(0))
36
37
38def roll(dice: int, times: int, faces: int = 6) -> Counter[int]:
39    """dice 個のダイスを times 回振り、和の出現回数を返す。"""
40    return Counter(sum(random.randint(1, faces) for _ in range(dice)) for _ in range(times))
41
42
43def show_theory() -> None:
44    for n in (1, 2, 3):
45        dist = sum_distribution(n)
46        print(f"{n}d6: 期待値 {float(mean(dist)):.2f}、分散 {float(variance(dist)):.3f}")
47        for v, p in sorted(dist.items()):
48            print(f"  {v:2d}: {float(p):.3f} {'#' * round(float(p) * 100)}")
49
50
51def compare(dice: int, times: int) -> None:
52    dist = sum_distribution(dice)
53    counts = roll(dice, times)
54    print(f"{dice}d6 を {times} 回振った結果（理論値との比較）")
55    for v, p in sorted(dist.items()):
56        print(f"  {v:2d}: {counts[v]:5d} 回  割合 {counts[v] / times:.3f}  理論値 {float(p):.3f}")
57
58
59def main() -> None:
60    if len(sys.argv) == 3:
61        compare(int(sys.argv[1]), int(sys.argv[2]))
62    else:
63        show_theory()
64
65
66if __name__ == "__main__":
67    main()

gacha.py

 1"""ガチャで全種類をそろえるまでの回数を計算・シミュレーションする（5 章）。
 2
 3どの種類も同じ確率で出るとき、n 種類をそろえるまでの回数の期待値は
 4n × (1 + 1/2 + ... + 1/n) になる（クーポンコレクター問題）。
 5
 6実行方法:
 7    py gacha.py          10 種類で計算する
 8    py gacha.py 30       種類の数を指定する
 9"""
10
11import random
12import statistics
13import sys
14
15
16def expected_draws(kinds: int) -> float:
17    """全種類をそろえるまでの回数の期待値を返す。"""
18    return kinds * sum(1 / k for k in range(1, kinds + 1))
19
20
21def draws_to_complete(kinds: int, rng: random.Random) -> int:
22    """1 回分のシミュレーション。全種類がそろうまでに引いた回数を返す。"""
23    owned: set[int] = set()
24    draws = 0
25    while len(owned) < kinds:
26        owned.add(rng.randrange(kinds))
27        draws += 1
28    return draws
29
30
31def simulate(kinds: int, trials: int, seed: int = 1) -> list[int]:
32    rng = random.Random(seed)
33    return [draws_to_complete(kinds, rng) for _ in range(trials)]
34
35
36def main() -> None:
37    kinds = int(sys.argv[1]) if len(sys.argv) > 1 else 10
38    results = sorted(simulate(kinds, 20_000))

shuffle.py

 1"""正しいシャッフルと誤ったシャッフルの偏りを比べる（5 章）。
 2
 33 枚のカード ABC の並べ方 6 通りが、それぞれ何回出るかを数える。
 4誤ったシャッフルは乱数の組合せが 3^3 = 27 通りで、6 で割り切れないため必ず偏る。
 5
 6実行方法: py shuffle.py
 7"""
 8
 9import itertools
10import random
11from collections import Counter
12
13
14def fisher_yates(cards: list[str], rng: random.Random) -> list[str]:
15    """正しいシャッフル。後ろから順に、まだ決まっていない位置の中から 1 枚を選んで入れ替える。"""
16    result = cards[:]
17    for i in range(len(result) - 1, 0, -1):
18        j = rng.randint(0, i)
19        result[i], result[j] = result[j], result[i]
20    return result
21
22
23def naive_shuffle(cards: list[str], rng: random.Random) -> list[str]:
24    """誤ったシャッフル。各位置を、全体の中から選んだ 1 枚と入れ替える。"""
25    result = cards[:]
26    for i in range(len(result)):
27        j = rng.randrange(len(result))
28        result[i], result[j] = result[j], result[i]
29    return result
30
31
32def naive_exact(cards: list[str]) -> Counter[str]:
33    """誤ったシャッフルの乱数の組合せ 27 通りをすべて試し、並べ方ごとの回数を返す。"""
34    counts: Counter[str] = Counter()
35    n = len(cards)
36    for choice in itertools.product(range(n), repeat=n):
37        result = cards[:]
38        for i, j in enumerate(choice):
39            result[i], result[j] = result[j], result[i]
40        counts["".join(result)] += 1
41    return counts
42
43
44def frequencies(trials: int, seed: int = 1) -> dict[str, tuple[float, float]]:
45    """並べ方ごとに、(正しいシャッフルの割合, 誤ったシャッフルの割合) を返す。"""
46    rng = random.Random(seed)
47    cards = ["A", "B", "C"]
48    good = Counter("".join(fisher_yates(cards, rng)) for _ in range(trials))
49    bad = Counter("".join(naive_shuffle(cards, rng)) for _ in range(trials))
50    orders = sorted("".join(p) for p in itertools.permutations(cards))
51    return {o: (good[o] / trials, bad[o] / trials) for o in orders}
52
53
54def main() -> None:
55    print("並べ方  正しい  誤り   （理論値 1/6 = 0.167）")
56    for order, (good, bad) in frequencies(60_000).items():
57        print(f"{order}     {good:.3f}   {bad:.3f}")
58    exact = naive_exact(["A", "B", "C"])
59    print("誤ったシャッフルの正確な回数（27 通り中）: " + "、".join(f"{k} {v}" for k, v in sorted(exact.items())))
60
61
62if __name__ == "__main__":
63    main()

13def expected_score(rating_a: float, rating_b: float) -> float:
14    """A が B に勝つ見込み（勝ちを 1、負けを 0 としたときの期待値）を返す。"""
15    return 1 / (1 + 10 ** ((rating_b - rating_a) / 400))

18def update(rating_a: float, rating_b: float, score_a: float) -> tuple[float, float]:
19    """試合結果 score_a（勝ち 1、引き分け 0.5、負け 0）から、新しいレーティングを返す。"""
20    delta = K * (score_a - expected_score(rating_a, rating_b))
21    return rating_a + delta, rating_b - delta

26    # 本当の強さ（プレイヤーには見えない）。この差から勝率を決める

30    for round_no in range(1, 101):
31        for i, a in enumerate(names):
32            for b in names[i + 1:]:
33                win = rng.random() < expected_score(true_strength[a], true_strength[b])
34                ratings[a], ratings[b] = update(ratings[a], ratings[b], 1.0 if win else 0.0)
35        if round_no in (1, 5, 20, 100):
36            print(f"{round_no:3d} 回戦後: " + "  ".join(f"{n} {r:6.0f}" for n, r in ratings.items()))

  1"""すごろくのルールを変えて、勝敗がどう決まるかを測る（2 章、8 章）。
  2
  3基本ルール:
  4    * 2 人が交互に 1 個のダイスを振り、出た目だけ進む。先にゴールに着いた（越えた）方が勝ち。
  5    * 1 回の手番で 1 回だけ振り直してよい。振り直したら 2 回目の目に従う。
  6
  7振り直しの判断が、このゲームで唯一の「プレイヤーの選択」である。
  8判断の仕方（戦略）として、次の 3 つを用意する。
  9    * skilled: 3 以下が出たら振り直す（期待値 3.5 より小さい目を捨てる）
 10    * random:  2 回に 1 回、でたらめに振り直す
 11    * never:   振り直さない

 13ルールの変更（関数 run の引数 rule）:
 14    * basic:    基本ルール
 15    * catch_up: 後ろにいるプレイヤーは出た目に 2 を足す（追いつきルール、負のフィードバック）
 16    * run_away: 前にいるプレイヤーは出た目に 2 を足す（独走ルール、正のフィードバック）
 17
 18実行方法:
 19    py sugoroku.py                   演習 3 と演習 4 の表をすべて表示する
 20    py sugoroku.py --play            基本ルールで遊ぶ（相手は skilled）
 21"""

 23import random
 24import statistics
 25import sys
 26from dataclasses import dataclass
 27from typing import Literal
 28
 29Strategy = Literal["skilled", "random", "never"]
 30Rule = Literal["basic", "catch_up", "run_away"]
 31RULES: list[Rule] = ["basic", "catch_up", "run_away"]
 32BONUS = 2

 35@dataclass
 36class GameResult:
 37    winner: int        # 勝ったプレイヤーの番号（0 が先手）
 38    turns: int         # 両者の手番の合計
 39    lead_changes: int  # 先頭が入れ替わった回数
 40    gap: int           # ゲーム終了時に負けた側がゴールまで残していたマス数
 41
 42
 43def wants_reroll(strategy: Strategy, face: int, rng: random.Random) -> bool:
 44    if strategy == "skilled":
 45        return face <= 3
 46    if strategy == "random":
 47        return rng.random() < 0.5
 48    return False
 49
 50
 51def bonus(rule: Rule, me: int, other: int) -> int:
 52    """ルールの変更による出目への加算を返す。"""
 53    if rule == "catch_up" and me < other:
 54        return BONUS
 55    if rule == "run_away" and me > other:
 56        return BONUS
 57    return 0
 58
 59
 60def play_game(rule: Rule, length: int, strategies: tuple[Strategy, Strategy], rng: random.Random) -> GameResult:
 61    """1 ゲームを最後まで進め、結果を返す。"""
 62    pos = [0, 0]
 63    leader: int | None = None
 64    lead_changes = 0
 65    turns = 0
 66    player = 0
 67    while True:
 68        face = rng.randint(1, 6)
 69        if wants_reroll(strategies[player], face, rng):
 70            face = rng.randint(1, 6)
 71        pos[player] += face + bonus(rule, pos[player], pos[1 - player])
 72        turns += 1
 73        if pos[player] >= length:
 74            return GameResult(player, turns, lead_changes, length - pos[1 - player])
 75        if pos[0] != pos[1]:
 76            new_leader = 0 if pos[0] > pos[1] else 1
 77            if leader is not None and new_leader != leader:
 78                lead_changes += 1
 79            leader = new_leader
 80        player = 1 - player
 81
 82
 83def run(rule: Rule, length: int, strategies: tuple[Strategy, Strategy], games: int,
 84        seed: int = 1, swap: bool = False) -> list[GameResult]:
 85    """games 回対戦する。swap が真なら先手と後手を 1 回ごとに入れ替え、結果の番号は strategies の順にそろえる。"""
 86    rng = random.Random(seed)
 87    results = []
 88    for g in range(games):
 89        if swap and g % 2 == 1:
 90            r = play_game(rule, length, (strategies[1], strategies[0]), rng)
 91            r.winner = 1 - r.winner
 92        else:
 93            r = play_game(rule, length, strategies, rng)
 94        results.append(r)
 95    return results
 96
 97
 98def win_rate(results: list[GameResult], player: int = 0) -> float:
 99    return sum(r.winner == player for r in results) / len(results)
